การเชื่อมต่อ Google Sheets ด้วย Google Apps Script (GAS)

คัดลอกโค้ดนี้ไปวางที่ Extensions > Apps Script ใน Google Sheets เพื่อรองรับการบันทึกข้อมูลแบบเรียลไทม์

คำสั่งสร้างตารางอัตโนมัติ (SetupSheet) & Web App Service: 
// =========================================================================
// GOOGLE APPS SCRIPT (GAS) FOR HOSPITAL FIRE EXTINGUISHER INSPECTION
// Form Code: FM-ENV-FIRE-01 (รพ.หนองสองห้อง)
// =========================================================================

function setupSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Inspection_Logs");
  if (!sheet) {
    sheet = ss.insertSheet("Inspection_Logs");
  }
  
  // Create Complete Headers Automatically
  var headers = [
    "Timestamp", "Asset_ID", "Department", "Location", "Extinguisher_Type", 
    "Capacity", "Fire_Class", "Next_Test_Date", "Inspect_Date", "Inspect_Period", 
    "Inspector_Name", "Overall_Status", "M01", "M02", "M03", "M04", 
    "M05", "M06", "M07", "M08", "M09", "M10", "M11", "M12", 
    "Defect_Severity", "Defect_Details", "Missing_Fields_Flag"
  ];
  
  sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
  sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold").setBackground("#dcfce7");
  sheet.setFrozenRows(1);
}

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("Inspection_Logs");
    if (!sheet) {
      setupSheet();
      sheet = ss.getSheetByName("Inspection_Logs");
    }
    
    sheet.appendRow([
      new Date(),
      data.asset_id || "",
      data.department || "",
      data.location || "",
      data.extinguisher_type || "",
      data.capacity || "",
      data.fire_class || "",
      data.next_test_date || "",
      data.inspect_date || "",
      data.inspect_period || "",
      data.inspector_name || "",
      data.overall_status || "",
      data.M01 || "", data.M02 || "", data.M03 || "", data.M04 || "",
      data.M05 || "", data.M06 || "", data.M07 || "", data.M08 || "",
      data.M09 || "", data.M10 || "", data.M11 || "", data.M12 || "",
      data.defect_severity || "",
      data.defect_details || "",
      data.has_missing ? "YES" : "NO"
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({result: "success"})).setMimeType(ContentService.MimeType.JSON);
  } catch(error) {
    return ContentService.createTextOutput(JSON.stringify({result: "error", message: error.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}
